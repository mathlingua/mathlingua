[\axiom]
Axiom:
given:
. X is \set
where:
. [logic.not]
  not:
  . x = y
then:
. [logic.if]
  if:
  . x = x
  then:
  . y = y
iff:
. [logic.have]
  have:
  . y = y
  iff:
  . x = x
Appendix:
. called: "Every element equals itself"
. [axiom.written]
  written:
  . "axiom"
. justification:
  . [axiom.justified]
    for: (:logic.if:)
    have:
    . x = x
    asserting:
    . x = x
. aliases:
  . [axiom.alias]
    alias: axiom(x_) :=> x
. references:
  . $axiom.ref
. metadata:
  . version: "1.0"
